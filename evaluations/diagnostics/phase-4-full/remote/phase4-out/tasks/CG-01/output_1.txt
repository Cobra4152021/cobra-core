```python
import argparse
import json
import sys

def validate_json_file(file_path):
    try:
        with open(file_path, 'r', encoding='utf-8') as file:
            data = json.load(file)
    except FileNotFoundError:
        print(f"Error: File not found: {file_path}", file=sys.stderr)
        return 1
    except json.JSONDecodeError:
        print(f"Error: Invalid JSON in file: {file_path}", file=sys.stderr)
        return 2

    required_keys = ['id', 'name', 'email']
    for key in required_keys:
        if key not in data:
            print(f"Error: Missing required key '{key}' in file: {file_path}", file=sys.stderr)
            return 3

    return 0

def main():
    parser = argparse.ArgumentParser(description='Validate JSON file for required keys.')
    parser.add_argument('file_path', type=str, help='Path to the JSON file')
    args = parser.parse_args()

    exit_code = validate_json_file(args.file_path)
    sys.exit(exit_code)

if __name__ == '__main__':
    main()
```

### Example Invocation:
```bash
python3.12 validate_json.py /path/to/data.json
```

### Exit Code Table:

| Exit Code | Description                          | Condition                                                                 |
|-----------|--------------------------------------|---------------------------------------------------------------------------|
| 0         | Success                               | All required keys (`id`, `name`, `email`) are present and JSON is valid |
| 1         | Missing file                          | The specified file does not exist                                        |
| 2         | Invalid JSON                          | The file exists but is not valid JSON                                    |
| 3         | Missing required keys                 | The file is valid JSON but is missing one or more required keys         |
