```python
def normalize_records(records):
    """
    Transforms a list of dict records by:
    - Normalizing date fields named "date" to ISO-8601 YYYY-MM-DD when parseable
    - Dropping records where email is null/None/missing
    - Deduping by id, keeping the first occurrence
    """
    seen_ids = set()
    result = []
    
    for record in records:
        # Skip if email is missing or None
        if 'email' not in record or record['email'] is None:
            continue
        
        # Normalize date field if present
        if 'date' in record:
            try:
                from datetime import datetime
                date_obj = datetime.strptime(record['date'], "%Y-%m-%d")
                record['date'] = date_obj.strftime("%Y-%m-%d")
            except ValueError:
                # If date is not in the expected format, leave it as-is
                pass
        
        # Dedupe by id
        record_id = record.get('id')
        if record_id and record_id not in seen_ids:
            seen_ids.add(record_id)
            result.append(record)
    
    return result
```
